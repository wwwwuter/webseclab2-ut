from flask import request
import sqlite3

@app.route('/api/user')
def get_user():
    user_id = request.args.get('id')
    conn = sqlite3.connect('app.db')
    # ✓ 使用参数化查询 (?)
    cursor = conn.execute(
        "SELECT * FROM users WHERE id = ?",
        (user_id,)
    )
    return jsonify(cursor.fetchone())

const express = require('express');
